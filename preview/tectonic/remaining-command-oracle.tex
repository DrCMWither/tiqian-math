\newcommand{\traceformula}[2]{%
  \message{TIQIAN-REMAINING-#1-BEGIN}%

}
\newcommand{\runoracles}[1]{%
  \traceformula{#1-ATOP}{{a\atop b}}%
  \traceformula{#1-CHOOSE}{{a\choose b}}%
  \traceformula{#1-CHOOSE-DISPLAY}{\displaystyle{a\choose b}}%
  \traceformula{#1-BF}{\bf{0}}%
  \traceformula{#1-BF-MIXED}{\bf{0+x+\alpha+\sum}}%
  \traceformula{#1-NOT}{\mu\not\equiv\mu}%
  \traceformula{#1-NOT-OVERLAY}{\not p}%
  \traceformula{#1-NOT-OVERLAY-REL-LEFT}{a}%
  \traceformula{#1-NOT-OVERLAY-REL-CONTROL}{a\propto b}%
  \traceformula{#1-NOT-OVERLAY-REL}{a\not\propto b}%
  \traceformula{#1-NOT-OVERLAY-REL-RIGHT}{b}%
  \traceformula{#1-CANCEL}{\cancel{x+1}}%
  \traceformula{#1-TEXTBF}{\textbf{1}}%
  \traceformula{#1-HLINE}{\begin{array}{c}a\\\hline b\end{array}}%
}
